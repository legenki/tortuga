# Route Map Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Show every app's Tor route on a dotted world map at the top of the macOS app's Route screen, and refresh the menu bar panel with app colours and an app count.

**Architecture:** Spec: `docs/superpowers/specs/2026-11-22-no-torrents-and-route-map-design.md`, step 2. Map data (land dots and country label points) is generated once from Natural Earth by a Node script and committed as `WorldMap.json` in TortugaKit's resources. Pure, tested logic lives in TortugaKit: `WorldMap` (loading), `MapProjection` (equirectangular fit), `AppColor` (stable colour index and dot spreading), `MapRoute` (what to draw for an `EngineState`). TortugaUI draws it with a SwiftUI `Canvas` in `RouteMap`, and wires highlight between the map and the app rows.

**Tech Stack:** Swift 6.2, SwiftUI (macOS 26), swift-testing, XcodeGen; Node 20+ for the one-off data script.

**Branch:** `feat/route-map`, cut from `main` after PR #8 (Remove torrents) is merged.

```js
// Writes TortugaKit's WorldMap.json: land as a 2° grid of dots, and each
// country's label point by ISO 3166 alpha-2 code. Data: Natural Earth
// (public domain), fetched from a pinned release of natural-earth-vector.
import { writeFileSync } from 'node:fs';
import { geoContains } from 'd3-geo';

const NE = 'https://raw.githubusercontent.com/nvkelso/natural-earth-vector/v5.1.2/geojson';
const OUT = new URL('../../Packages/TortugaKit/Sources/TortugaKit/Resources/WorldMap.json', import.meta.url);
const STEP = 2;
// Antarctica is left out: no relays there, and it would take a third of the map.
const SOUTH = -58;
const NORTH = 84;

async function get(name) {
  const res = await fetch(`${NE}/${name}`);
  if (!res.ok) throw new Error(`${name}: HTTP ${res.status}`);
  return res.json();
}

const land = await get('ne_110m_land.geojson');
const countries = await get('ne_50m_admin_0_countries.geojson');

const dots = [];
for (let lat = NORTH - STEP / 2; lat >= SOUTH; lat -= STEP) {
  for (let lon = -180 + STEP / 2; lon < 180; lon += STEP) {
    if (land.features.some(f => geoContains(f, [lon, lat]))) dots.push([lon, lat]);
  }
}

const places = {};
for (const f of countries.features) {
  const p = f.properties;
  const code = p.ISO_A2_EH && p.ISO_A2_EH !== '-99' ? p.ISO_A2_EH : p.ISO_A2;
  if (!code || code === '-99' || places[code]) continue;
  places[code] = [Math.round(p.LABEL_X * 100) / 100, Math.round(p.LABEL_Y * 100) / 100];
}

writeFileSync(OUT, JSON.stringify({ dots, countries: places }) + '\n');
console.log(`${dots.length} dots, ${Object.keys(places).length} countries → ${OUT.pathname}`);
```

`WorldMap.swift`:

```swift
import Foundation

/// A point on Earth in degrees.
public struct Coordinate: Sendable, Hashable, Codable {
    public var lon: Double
    public var lat: Double

    public init(lon: Double, lat: Double) {
        self.lon = lon
        self.lat = lat
    }
}

/// The land as dots and where each country is, from the bundled
/// WorldMap.json (Natural Earth; see macos/tools/map-data).
public struct WorldMap: Sendable {
    public let dots: [Coordinate]
    private let countries: [String: Coordinate]

    public static let empty = WorldMap(dots: [], countries: [:])
    public static let shared = load(from: Bundle.module.url(forResource: "WorldMap", withExtension: "json"))

    init(dots: [Coordinate], countries: [String: Coordinate]) {
        self.dots = dots
        self.countries = countries
    }

    /// Where a country is drawn: its label point. nil for an unknown code.
    public func place(country code: String) -> Coordinate? {
        countries[code.uppercased()]
    }

    /// The map in `url`, or an empty one if it is missing or unreadable: the
    /// routes are still drawn, just without land.
    static func load(from url: URL?) -> WorldMap {
        struct File: Decodable {
            var dots: [[Double]]
            var countries: [String: [Double]]
        }
        guard let url, let data = try? Data(contentsOf: url),
              let file = try? JSONDecoder().decode(File.self, from: data) else { return .empty }
        let point = { (p: [Double]) in p.count == 2 ? Coordinate(lon: p[0], lat: p[1]) : nil }
        return WorldMap(dots: file.dots.compactMap(point), countries: file.countries.compactMapValues(point))
    }
}
```

```swift
import CoreGraphics

/// Equirectangular projection fitted to a set of places: degrees of longitude
/// and latitude are the same length, the places fill the view inside the
/// padding, and at least `minSpan` degrees of longitude show.
public struct MapProjection: Sendable, Equatable {
    /// The whole map: Natural Earth land without Antarctica (see map-data).
    public static let world = (lon: -180.0...180.0, lat: -58.0...84.0)

    let scale: Double
    let originLon: Double
    let originLat: Double
    let offset: CGPoint

    public init(fitting places: [Coordinate], in size: CGSize, padding: CGFloat, minSpan: Double = 40) {
        var lon = Self.world.lon
        var lat = Self.world.lat
        if let first = places.first {
            lon = places.reduce(first.lon...first.lon) { min($0.lowerBound, $1.lon)...max($0.upperBound, $1.lon) }
            lat = places.reduce(first.lat...first.lat) { min($0.lowerBound, $1.lat)...max($0.upperBound, $1.lat) }
        }
        let width = max(Double(size.width - 2 * padding), 1)
        let height = max(Double(size.height - 2 * padding), 1)
        // Widen to the minimum span, and to the view's shape, around the centre.
        let lonSpan = max(lon.upperBound - lon.lowerBound, places.isEmpty ? 0 : minSpan)
        let latSpan = max(lat.upperBound - lat.lowerBound, lonSpan * height / width)
        let lonSpanFitted = max(lonSpan, latSpan * width / height)
        let midLon = (lon.lowerBound + lon.upperBound) / 2
        let midLat = (lat.lowerBound + lat.upperBound) / 2
        scale = min(width / lonSpanFitted, height / latSpan)
        originLon = midLon - lonSpanFitted / 2
        originLat = midLat + latSpan / 2
        offset = CGPoint(x: padding + (width - lonSpanFitted * scale) / 2,
                         y: padding + (height - latSpan * scale) / 2)
    }

    public func point(_ c: Coordinate) -> CGPoint {
        CGPoint(x: offset.x + (c.lon - originLon) * scale, y: offset.y + (originLat - c.lat) * scale)
    }

    public func coordinate(at p: CGPoint) -> Coordinate {
        Coordinate(lon: originLon + (p.x - offset.x) / scale, lat: originLat - (p.y - offset.y) / scale)
    }
}
```

```swift
import CoreGraphics

/// Which of the palette's colours an app's route is drawn in, and how dots
/// that share a place are spread apart. The palette itself is in TortugaUI.
public enum AppColor {
    public static let count = 8

    /// The same for an app on every run: FNV-1a of its identifier, not
    /// Swift's `hashValue`, which is seeded per process.
    public static func index(for appID: String) -> Int {
        Int(fnv1a(appID) % UInt32(count))
    }

    static func fnv1a(_ s: String) -> UInt32 {
        var hash: UInt32 = 0x811C9DC5
        for byte in s.utf8 {
            hash ^= UInt32(byte)
            hash = hash &* 0x0100_0193
        }
        return hash
    }

    /// Where the n-th dot on a shared place goes, in points: the first stays,
    /// the others sit 6 points away at golden-angle steps.
    public static func offset(slot: Int) -> CGVector {
        guard slot > 0 else { return .zero }
        let angle = Double(slot) * 137.50776 * .pi / 180
        return CGVector(dx: 6 * cos(angle), dy: 6 * sin(angle))
    }
}
```

`MapRoute.swift`:

```swift
import Foundation

/// What the route map draws for a state: the trunk from this Mac to the guard,
/// and each routed app's tail through its middle to its exit. Places come from
/// the countries Tor reports; a country the map doesn't know is skipped.
public struct MapRoute: Sendable, Equatable {
    public struct Tail: Sendable, Equatable, Identifiable {
        public var app: AppID
        /// Guard (or the Mac, without one), middle, exit: the known ones.
        public var points: [Coordinate]
        public var exit: Coordinate?
        public var colorIndex: Int
        /// 0 for the first app (by identifier) whose exit is in that country,
        /// then 1, 2…: how far to spread its dot (AppColor.offset).
        public var exitSlot: Int
        public var id: String { app.id }
    }

    public var mac: Coordinate?
    public var guardPlace: Coordinate?
    public var bridge: BridgeKind?
    public var tails: [Tail]

    /// `region`: the Mac's region code (Locale.current.region), if any.
    public init(state: EngineState, region: String?, map: WorldMap = .shared) {
        mac = region.flatMap(map.place(country:))
        guardPlace = state.trunk.guardRelay.flatMap { map.place(country: $0.country) }
        bridge = state.trunk.bridge
        let start = guardPlace ?? mac
        let routed = state.apps
            .filter { !state.exceptions.contains($0.app) }
            .sorted { $0.app.id < $1.app.id }
        var taken: [String: Int] = [:]
        tails = routed.map { route in
            let exitCode = route.exit.country.uppercased()
            let slot = taken[exitCode, default: 0]
            taken[exitCode] = slot + 1
            let middle = map.place(country: route.middle.country)
            let exit = map.place(country: route.exit.country)
            return Tail(app: route.app, points: [start, middle, exit].compactMap { $0 }, exit: exit,
                        colorIndex: AppColor.index(for: route.app.id), exitSlot: slot)
        }
    }

    /// Every place drawn, for fitting the map to them.
    public var places: [Coordinate] {
        [mac, guardPlace].compactMap { $0 } + tails.flatMap(\.points)
    }
```

```swift
/// The route map in words, for VoiceOver.
enum RouteSummary {
```

```swift
import AppKit
import SwiftUI
import TortugaKit

/// The world as dots, with this Mac's trunk and each app's route drawn on it.
/// `highlighted`: the app drawn at full strength (the others fade); nil: all.
struct RouteMap: View {
    let state: EngineState
    var highlighted: String?
    var region: String? = Locale.current.region?.identifier
    var onSelect: (String) -> Void = { _ in }

    private var route: MapRoute { MapRoute(state: state, region: region) }
    private var lit: Bool { state.status == .on }

    var body: some View {
        let route = route
        GeometryReader { geo in
            let projection = MapProjection(fitting: route.places, in: geo.size, padding: 18)
            ZStack {
                Canvas { context, _ in draw(route, projection, in: &context) }
                ForEach(route.tails) { tail in
                    if let exit = tail.exit {
                        let p = projection.point(exit) + AppColor.offset(slot: tail.exitSlot)
                        Circle().fill(.clear).contentShape(.circle).frame(width: 16, height: 16)
                            .position(p)
                            .onTapGesture { onSelect(tail.id) }
                            .accessibilityHidden(true)
                    }
                }
            }
        }
        .accessibilityElement(children: .ignore)
        .accessibilityLabel(RouteSummary.text(for: state, name: AppInfo.name))
        .accessibilityIdentifier("route.map")
    }

    private func draw(_ route: MapRoute, _ projection: MapProjection, in context: inout GraphicsContext) {
        let land = Color.secondary.opacity(0.28)
        for dot in WorldMap.shared.dots {
            let p = projection.point(dot)
            context.fill(Path(ellipseIn: CGRect(x: p.x - 1, y: p.y - 1, width: 2, height: 2)), with: .color(land))
        }
        guard state.mode == .tor else { return }

        if let mac = route.mac, let guardPlace = route.guardPlace {
            let color: Color = !lit ? .secondary.opacity(0.5) : route.bridge == nil ? .route : .bridge
            let style = route.bridge == .snowflake
                ? StrokeStyle(lineWidth: 2.5, lineCap: .round, dash: [1, 5])
                : StrokeStyle(lineWidth: 3, lineCap: .round)
            context.stroke(curve(projection.point(mac), projection.point(guardPlace)), with: .color(color), style: style)
        }

        for tail in route.tails {
            let faded = highlighted != nil && highlighted != tail.id
            let color = (lit ? Color.app(tail.id) : .secondary).opacity(faded ? 0.3 : 1)
            let points = tail.points.map(projection.point)
            let end = points.last.map { $0 + AppColor.offset(slot: tail.exitSlot) }
            var path = Path()
            for (i, p) in points.enumerated().dropFirst() {
                let to = i == points.count - 1 ? (end ?? p) : p
                path.addPath(curve(points[i - 1], to))
            }
            context.stroke(path, with: .color(color), style: StrokeStyle(lineWidth: highlighted == tail.id ? 2.5 : 1.8, lineCap: .round))
            if points.count == 3 { ring(points[1], color, 3, in: &context) }
            if let end {
                context.fill(Path(ellipseIn: CGRect(x: end.x - 4, y: end.y - 4, width: 8, height: 8)), with: .color(color))
            }
        }

        if let mac = route.mac { ring(projection.point(mac), .primary, 4, in: &context) }
        if let guardPlace = route.guardPlace { ring(projection.point(guardPlace), lit ? .route : .secondary, 4, in: &context) }
    }

    /// A gentle arc from a to b: the control point is pushed sideways by an
    /// eighth of the distance.
    private func curve(_ a: CGPoint, _ b: CGPoint) -> Path {
        var path = Path()
        path.move(to: a)
        let mid = CGPoint(x: (a.x + b.x) / 2, y: (a.y + b.y) / 2)
        let control = CGPoint(x: mid.x + (b.y - a.y) / 8, y: mid.y - (b.x - a.x) / 8)
        path.addQuadCurve(to: b, control: control)
        return path
    }

    private func ring(_ p: CGPoint, _ color: Color, _ r: CGFloat, in context: inout GraphicsContext) {
        let circle = Path(ellipseIn: CGRect(x: p.x - r, y: p.y - r, width: 2 * r, height: 2 * r))
        context.fill(circle, with: .color(Color(nsColor: .windowBackgroundColor)))
        context.stroke(circle, with: .color(color), lineWidth: 2)
    }
}

private func + (p: CGPoint, v: CGVector) -> CGPoint { CGPoint(x: p.x + v.dx, y: p.y + v.dy) }
```
